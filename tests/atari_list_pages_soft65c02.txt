; Each scenario primes the result sentinel and independently checks wire bytes,
; status, reply fields, and whether the raw call happened.

marker $$plain request and response$$
memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
registers flush
memory write $_test_case 0x(00)
memory write $_test_seen 0x(FF)
memory write $_test_result 0x(FF)
run $init_harness until CP = $profile_start
registers show cycle_count $$case 0 start$$
run until CP = $profile_done
registers show cycle_count $$case 0 end$$
run until CP = $end_init_harness
assert $_test_seen = 0x00 $$selected scenario ran$$
assert $_test_result = 0x00 $$plain request and response$$
